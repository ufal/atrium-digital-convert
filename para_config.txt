[components]
# name = <license> ; <always|conditional> ; <role>
# "always"      -> listed in the `components` of every run's paradata
# "conditional" -> listed only when the run calls log_component() (or, via the CLI
#                  shim, when `--component NAME` is passed to `start`/`success`).
# Paradata 3.0 list each component a run used with these terms, as facts: no licence is
# computed or merged from them (D2, atrium-project#84).
#
# ── the born-digital converter (program `digital-convert`) ─────────────────────
# Without these rows log_component("pdfplumber") would list the component with the terms
# "UNKNOWN".
#
# All "conditional": a run only touches the libraries the input format needs, and
# api_util/digital_to_json.record_run() logs what it used, with versions. The light
# engine's rows are the libraries that parse the file (pdfminer.six under pdfplumber;
# pypdfium2 for page labels and the OCR-layer census; lxml under python-docx and the
# shared reader) plus jsonschema, which gates every docproc.
pdfplumber   = MIT          ; conditional ; born-digital PDF words, geometry, fonts, ruled tables
pdfminer.six = MIT          ; conditional ; PDF parser under pdfplumber
pypdfium2    = Apache-2.0   ; conditional ; PDF page labels + text render mode (OCR-layer census), page counts
python-docx  = MIT          ; conditional ; DOCX package, table grids
lxml         = BSD-3-Clause ; conditional ; DOCX WordprocessingML walk; ODT/ODS/XLSX parts in the shared reader
jsonschema   = MIT          ; conditional ; digital-convert's output gate (validate_document)
#
# The shared reader (atrium-digital-convert#2 W2): text_formats.py, vendored byte-identical
# from atrium-ocr-postprocess (MIT, as this repository), reads ODT, ODS, XLSX and RTF, and decodes
# the text-only inputs (TXT, CSV, TSV, JSON; atrium-digital-convert#9).
text_formats = MIT          ; conditional ; ODT/ODS/XLSX/RTF reader, text decoding (vendored from atrium-ocr-postprocess)
#
# Headless LibreOffice (#2 W2; accepted by AMČR on #4, 2026-09-26): converts a legacy DOC/XLS to
# DOCX/XLSX before the readers above read it, and since atrium-digital-convert#9 renders an office
# document (DOCX, DOC, ODT, RTF) to PDF for its pages; the text stays the native readers'. AMČR's
# condition on #4 was that it "adds nothing to the licence of the record": paradata 3.0 computes no
# licence at all (D2, atrium-project#84), so it is LOGGED like every component a run uses, with
# its version, as a fact (api_util/digital_to_json.record_run).
libreoffice  = MPL 2.0      ; conditional ; DOC -> DOCX and XLS -> XLSX conversion; office documents rendered to PDF for their pages (api_util/digital_legacy.py, api_util/office_render.py)
#
# Heavy engine (`--engine docling`, requirements_digital_docling.txt; PDF only; a local
# build, never in the service image). The two model rows are the weights Docling downloads,
# licensed separately from its MIT code. TableFormer's weights are CDLA-Permissive-2.0,
# which places no terms on Results, the output of a computation over the data.
docling              = MIT                 ; conditional ; heavy PDF engine: layout + table structure
docling-parse        = MIT                 ; conditional ; Docling's PDF backend
docling-layout-heron = Apache-2.0          ; conditional ; layout model weights (docling-project/docling-layout-heron)
docling-tableformer  = CDLA-Permissive-2.0 ; conditional ; table-structure model weights (docling-project/docling-models)
#
# ── flexiconv (optional; api_util/flexiconv_convert.py, vendored from nlp-enrich) ─
# MIT: its LICENSE file (since v0.3.11) and, from the pinned v0.3.12, its pyproject.toml say so, and
# `pip show` reports MIT (atrium-project#6, ufal/flexiconv#1). It asks to be cited: its CITATION.cff.
# Conditional: it is listed only for runs that actually convert documents with it -- and no stage
# of this repo does (the adapter is vendored for its tests and for direct use; nothing here calls
# log_component("flexiconv")), so the row never enters a run's component list here. Converting is
# nlp-enrich's api_flexiconv.sh, which logs it.
flexiconv   = MIT     ; conditional ; flexiconv document -> TEITOK converter (api_util/flexiconv_convert.py)
# (#8, atrium-project#83) The decision models of the offline tools/decide_documents.py, served at
# POST /v1/systemone and called through decision_client.py (vendored from page-classification): Clef,
# Clef-Flash (Cloudflare) and Strands Decider 2B, all Apache-2.0. No image and no service run them.
decision_model = Apache-2.0 ; conditional ; decision models (Clef, Clef-Flash, Strands Decider 2B) for the #8 experiments

# PyMuPDF is intentionally NOT listed and not used: it is AGPL-3.0, and every born-digital run
# would list it. See requirements_digital.txt before adding it.
