[components]
# name = <license> ; <always|conditional> ; <role>
# "always"      -> component is part of every core run (worst-case baseline)
# "conditional" -> only counts toward the license when log_component() is called
#                  (or, via the CLI shim, when `--component NAME` is passed to
#                  `start`/`success`).
#
# ── the born-digital converter (program `digital-convert`) ─────────────────────
# It is not bookkeeping. Without these rows log_component("pdfplumber") stores
# "UNKNOWN", and para_licenses.resolve_effective_license() treats UNKNOWN as
# MAXIMALLY RESTRICTIVE, so a born-digital record would come out with
# `provenance.license = "UNKNOWN"` — worse than the conservative CC BY-NC 4.0
# default, and the opposite of the permissive result the pdfplumber-over-PyMuPDF
# choice in requirements_digital.txt exists to secure.
#
# All "conditional": a run only touches the libraries the input format needs.
# api_util/digital_to_json.py LOGS what it uses (license_detail_for()). The light
# engine's rows are the libraries that parse the file (pdfminer.six under pdfplumber;
# pypdfium2 for page labels and the OCR-layer census; lxml under python-docx and the
# shared reader) plus jsonschema, which gates every record.
pdfplumber   = MIT          ; conditional ; born-digital PDF words, geometry, fonts, ruled tables
pdfminer.six = MIT          ; conditional ; PDF parser under pdfplumber
pypdfium2    = Apache-2.0   ; conditional ; PDF page labels + text render mode (OCR-layer census), page counts
python-docx  = MIT          ; conditional ; DOCX package, table grids
lxml         = BSD-3-Clause ; conditional ; DOCX WordprocessingML walk; ODT/ODS/XLSX parts in the shared reader
jsonschema   = MIT          ; conditional ; digital-convert's output gate (validate_document)
#
# The shared reader (atrium-digital-convert#2 W2): text_formats.py, vendored byte-identical
# from atrium-ocr-postprocess (MIT, as this repository), reads ODT, ODS, XLSX and RTF.
text_formats = MIT          ; conditional ; ODT/ODS/XLSX/RTF reader (vendored from atrium-ocr-postprocess)
#
# Legacy DOC and XLS (#2 W2; accepted by AMČR on #4, 2026-09-26). Headless LibreOffice
# converts the file to DOCX/XLSX before the readers above read it. DECLARED HERE, NOT
# LOGGED: AMČR's condition was that it "only converts the file, so it adds nothing to the
# licence of the record" — it changes the container format of a text it does not author —
# and log_component() would put MPL 2.0 (share-alike in para_licenses) into the record's
# licence union. The conversion is reported beside the record instead (the service's
# `reader.conversion`; DigitalDocument.conversion). Same treatment as flexiconv below.
libreoffice  = MPL 2.0      ; conditional ; DOC -> DOCX and XLS -> XLSX conversion only (api_util/digital_legacy.py; never logged)
#
# Heavy engine (`--engine docling`, requirements_digital_docling.txt; PDF only; a local
# build, never in the service image). The two model rows are the weights Docling downloads,
# licensed separately from its MIT code. TableFormer's weights are CDLA-Permissive-2.0,
# which para_licenses.py ranks with MIT: it places no terms on Results, the output of a
# computation over the data. So a docling run's record resolves to MIT.
docling              = MIT                 ; conditional ; heavy PDF engine: layout + table structure
docling-parse        = MIT                 ; conditional ; Docling's PDF backend
docling-layout-heron = Apache-2.0          ; conditional ; layout model weights (docling-project/docling-layout-heron)
docling-tableformer  = CDLA-Permissive-2.0 ; conditional ; table-structure model weights (docling-project/docling-models)
#
# ── flexiconv (optional; api_util/flexiconv_convert.py, vendored from nlp-enrich) ─
# Declares GPL-3.0-or-later in its pyproject.toml (checked at the pinned v0.3.10); para_licenses
# folds "-or-later" into GPL-3.0. Conditional: it only constrains runs that actually convert
# documents with it -- and no stage of this repo does (the adapter is vendored for its tests and
# for direct use; nothing here calls log_component("flexiconv")), so the row never enters a
# run's licence union here. Converting is nlp-enrich's api_flexiconv.sh, which logs it.
flexiconv   = GPL-3.0 ; conditional ; flexiconv document -> TEITOK converter (api_util/flexiconv_convert.py)
#
# PyMuPDF is intentionally NOT listed. It is AGPL-3.0, and rule 5's most-restrictive union
# would put AGPL-3.0 into provenance.license for every born-digital document. See
# requirements_digital.txt before adding it.
#
# The LLM/vocabulary rows (amcr_vocab, teater_data and the model TODO) left with the keyword
# code for atrium-keyword-extract (atrium-digital-convert#1).
